Step 3: Inspect Bombs on Grid

// Find all bombs on current level
setTimeout(() => {
    console.log('=== Bomb Inspection ===');
    let found = 0;
    for (let row = 0; row < game.grid.length; row++) {
        for (let col = 0; col < game.grid[row].length; col++) {
            const tile = game.grid[row][col];
            if (tile && tile.isBomb()) {
                found++;
                console.log(`💣 Bomb ${found}:`);
                console.log(`   Position: Row ${row}, Col ${col}`);
                console.log(`   Timer: ${tile.bombTimer}`);
                console.log(`   Value: ${tile.value}`);
            }
        }
    }
    console.log(`Total: ${found} bomb(s) found`);
}, 1000);

// Display visual grid in console
setTimeout(() => {
    console.log('=== Current Grid ===');
    for (let row = 0; row < game.grid.length; row++) {
        let line = '';
        for (let col = 0; col < game.grid[row].length; col++) {
            const tile = game.grid[row][col];
            if (tile) {
                if (tile.isBomb()) {
                    line += `[💣${tile.bombTimer}]`.padEnd(7);
                } else if (tile.blocker === 'stone') {
                    line += '[STONE]'.padEnd(7);
                } else if (tile.blocker === 'ice') {
                    line += `[ICE${tile.blockerDuration}]`.padEnd(7);
                } else {
                    line += `[${tile.value}]`.padEnd(7);
                }
            } else {
                line += '[ - ]  ';
            }
        }
        console.log(line);
    }
}, 1000);

Expected Results

✅ PASS Criteria:
• Bomb tile shows bomb icon (💣) and timer
• When merged in chain, bomb disappears completely
• Tile at bomb position shows merged value, no bomb indicator
• Other bombs not in chain have timer decrease by 1
❌ FAIL Criteria:
• Bomb remains after being merged
• Bomb timer doesn't decrease when not merged
• Game crashes when merging bomb
• Bomb doesn't trigger game over when timer reaches 0
